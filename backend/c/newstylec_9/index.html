9.1 常见的位图文件格式

第8章介绍过图像传感器感应每一个像素点上的光并以R、G、B三种颜色分量描述像素点的颜色，一幅图像由多个像素组成，这种用像素表达图像的图形被称为位图。

9.1.1 常见的位图格式

针对不同的场景和应用需求，位图目前存在多种格式，其中较常用的格式有JPG和PNG。各种图形格式的特点和相互关系如下：

  • RAW格式，原始图像文件，是数码相机从图像传感器上获取的未经压缩的原始数据，会记录相应拍摄所产生的一些原始数据，如快门设置、光圈值等信息，图像细节丰富，但占用空间较大。
  • JPG格式，使用最广泛的照片格式，在日常生活中，手机、数码相机拍摄的都是JPG格式的，其进行了压缩，虽然压缩会对图像质量进行破坏，但人眼难以分辨。
  • GIF格式，只支持256种颜色，文件较小，如常见的动图就是GIF格式。但由于专利限制，后来很快被PNG格式取代。
  • PNG格式，便携式网络图形，是一种无损压缩的位图图形格式。广泛用于互联网。
  • BMP格式，位图，是一种不压缩像素数据的文件格式。它逐个地描述每个像素点的颜色，这一点和RAW格式类似，但比RAW格式保存的信息少，文件占用空间较RAW小，但比JPG、PNG大。BMP文件支持不同的颜色位数，本章只讲解24位真彩色模式的BMP图像的显示，以降低程序复杂度。
9.1.2 BMP文件基本结构

BMP文件都遵循统一的文件格式规范，BMP文件的主要内容是像素数据，但除此之外BMP文件中还包含必备的图片信息，例如图片的宽度和高度，这些信息一般放在文件的开始部分，被称为文件头。

  • 文件头的大小一般是固定的，例如BMP文件的文件头为固定的54个字节（在只讨论24位真彩色、无压缩、标准40字节信息头模式的BMP图像的前提下），文件头里面除了存储图像的宽度、高度。还包含其他一些信息，如文件大小、像素数据的起始位置和颜色数等。
  • 在文件头之后是图片的像素数据，像素数据的多少取决于图片的大小和颜色模式，图片越大、包含的颜色数越多，像素数据就越多。

import struct

file1 = r'D:\\BC101\\Resources\\L09\\640_480_RED.bmp'


def parse_bmp_header(path):
    with open(path, 'rb') as f:
        header = f.read(54)
    print(f"\n==== {path} ====")
    print("原始54字节:", header)
    # 小端解析 BITMAPFILEHEADER + BITMAPINFOHEADER
    (sig, file_size, reserved, data_offset,
     ih_size, w, h, planes,
     bitcnt, comp, img_size, xppm,
     yppm, clrused, clrimp) = struct.unpack(
        '<2sI4sIIiiHHIIIIII', header)
    print(f"签名:{sig}, 文件大小:{file_size},像素偏移:{data_offset}")
    print(f"宽={w},高={h},位深={bitcnt},压缩={comp},像素区大小={img_size}")

9.2.1 打开BMP文件并读取文件头

要在程序中显示BMP文件，首先要打开文件并从文件头中获取一些有用的信息（图片的宽度、高度和允许的颜色数等），BMP文件的文件头占用文件开始的54个字节，这54字节又分为三个部分。

  • BMP文件标识（2字节），第1-2字节是BMP文件的特殊标识，分别是字节B和M，如果一个文件的第1、2字节不是这两个字母，则可以认为不是一个BMP文件，可以放弃下一步读取。
  • 位图文件头（12字节），第3-14字节被称为位图文件头，其中主要包含两项信息：BMP文件的大小和像素数据在文件中的起始位置。
  • 位图详细信息（40字节），第15-54字节包含位图文件的详细信息，其中我们关心的信息包括位图宽度（像素）和位图高度（像素）。

/// L09_01_READ_BMP_FILE.cpp
#ifdef _MSC_VER
// VS 环境生效
#define _CRT_SECURE_NO_WARNINGS
#pragma warning(disable : 6031)
#pragma comment(lib, "EasyXw.lib")
#endif

#include <stdio.h>
#include <conio.h>
#include <graphics.h>

int main()
{
    FILE *bmpfile = fopen("D:/BC101/Resources/L09/640_480_RED.bmp", "rb");
    if (bmpfile == NULL)
    {
        printf("打开BMP文件失败\n");
        return 1;
    }
    printf("打开BMP文件成功\n");
    // 获取标识信息
    char tag[2];
    // fread(缓冲区,每个元素大小,元素数量,文件)
    size_t ret_1 = fread(tag, 2, 1, bmpfile);
    if (ret_1 != 1)
    {
        printf("读取BMP标识失败，文件过短或损坏\n");
        fclose(bmpfile);
        return 2;
    }
    char first = tag[0];
    char second = tag[1];
    printf("%c%c\n", first, second);
    if (first != 'B' || second != 'M')
    {
        printf("打开的文件不是BMP文件格式\n");
        fclose(bmpfile);
        return 3;
    }
    printf("读取BMP文件标识成功\n");

    // 开始处理位图文件图信息
    // 第1-4字节，整型， BMP文件大小
    unsigned int fileSize = 0;
    size_t ret_2 = fread(&fileSize, 4, 1, bmpfile);
    if (ret_2 != 1)
    {
        printf("读取BMP文件大小异常\n");
        fclose(bmpfile);
        return 4;
    }
    printf("BMP文件大小: %u\n", fileSize);

    // 第5-8字节，保留区， 9-12字节，像素偏移量
    unsigned int reserved = 0;
    unsigned int offBits = 0;
    size_t ret_3 = fread(&reserved, 4, 1, bmpfile);
    size_t ret_4 = fread(&offBits, 4, 1, bmpfile);
    if (ret_3 != 1 || ret_4 != 1)
    {
        printf("读取文件头保留/偏移失败\n");
        fclose(bmpfile);
        return 5;
    }
    printf("像素偏移量: %u\n", offBits);

    // ========== BITMAPINFOHEADER 信息头 ==========
    unsigned int infoHeaderSize; // 信息头结构体长度，标准BMP为40
    int width;                   // 图像宽度，像素
    int height;                  // 图像高度，像素；>0像素行倒存，<0正向存储
    unsigned short planes;       // 色彩平面数，BMP固定等于1
    unsigned short bitCount;     // 每个像素位深；24代表24位真彩色
    unsigned int compression;    // 压缩方式；0=不压缩
    unsigned int imageSize;      // 像素数据总字节（包含行padding填充，可为0）
    unsigned int xPelsPerMeter;  // 水平分辨率：每米多少像素
    unsigned int yPelsPerMeter;  // 垂直分辨率：每米多少像素
    unsigned int clrUsed;        // 使用的调色板颜色数量，24位真彩色为0
    unsigned int clrImportant;   // 重要颜色数量，一般为0

    size_t ret;

    ret = fread(&infoHeaderSize, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读infoHeaderSize失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&width, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读width失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&height, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读height失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&planes, 2, 1, bmpfile);
    if (ret != 1)
    {
        printf("读planes失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&bitCount, 2, 1, bmpfile);
    if (ret != 1)
    {
        printf("读bitCount失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&compression, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读compression失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&imageSize, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读imageSize失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&xPelsPerMeter, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读xPelsPerMeter失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&yPelsPerMeter, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读yPelsPerMeter失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&clrUsed, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读clrUsed失败\n");
        fclose(bmpfile);
        return 6;
    }

    ret = fread(&clrImportant, 4, 1, bmpfile);
    if (ret != 1)
    {
        printf("读clrImportant失败\n");
        fclose(bmpfile);
        return 6;
    }

    printf("==== BITMAPINFOHEADER ====\n");
    printf("信息头大小: %u\n", infoHeaderSize);
    printf("图像宽: %d\n", width);
    printf("图像高: %d\n", height);
    printf("色彩平面: %u\n", planes);
    printf("位深: %u\n", bitCount);
    printf("压缩方式: %u (0=无压缩)\n", compression);
    printf("像素数据总大小(含padding): %u\n", imageSize);
    printf("X分辨率(每米像素): %u\n", xPelsPerMeter);
    printf("Y分辨率(每米像素): %u\n", yPelsPerMeter);
    printf("调色板颜色数量: %u\n", clrUsed);
    printf("重要颜色数量: %u\n", clrImportant);

    int bytes_per_pixel = bitCount / 8;                  // 每个像素占多少字节，24位就是3字节
    int line_raw_bytes = width * bytes_per_pixel;        // 一行原始像素字节，不含padding填充
    int line_padded_bytes = (line_raw_bytes + 3) & (~3); // 4字节对齐后的真实一行大小，包含padding
    printf("每行原始字节：%d, 对齐后实际行大小:%d\n", line_raw_bytes, line_padded_bytes);

    //跳到像素数据真正起始位置，不要硬编码54
    fseek(bmpfile, offBits, SEEK_SET);
    printf("已经fseek跳到像素偏移：%u\n", offBits);

    fclose(bmpfile);

    return 0;
}

struct BITMAP_HEADER
{
    unsigned int Size;        // 第1-4字节，无符号整型， BMP文件大小
    unsigned short Reserved1; // 第5-6字节，无符号短整型，保留区1
    unsigned short Reserved2; // 第7-8字节，无符号短整型，保留区2
    unsigned int OffBits;     // 第9-12字节，无符号整型， 像素偏移量
};

typedef struct BITMAP_HEADER BitmapHeader;

/// L09_01_READ_BMP_FILE.cpp
#ifdef _MSC_VER
// VS 环境生效
#define _CRT_SECURE_NO_WARNINGS
#pragma warning(disable : 6031)
#pragma comment(lib, "EasyXw.lib")

#include <stdio.h>
#include <conio.h>
#include <graphics.h>

// 位图文件头
struct BITMAP_HEADER
{
    unsigned int Size;        // 第1-4字节，无符号整型， BMP文件大小
    unsigned short Reserved1; // 第5-6字节，无符号短整型，保留区1
    unsigned short Reserved2; // 第7-8字节，无符号短整型，保留区2
    unsigned int OffBits;     // 第9-12字节，无符号整型， 像素偏移量
};

typedef struct BITMAP_HEADER BitmapHeader;

// 位图详细信息
struct BITMAP_INFORMATION
{
    // ========== BITMAPINFOHEADER 信息头 ==========
    unsigned int InfoHeaderSize; // 信息头结构体长度，标准BMP为40
    int Width;                   // 图像宽度，像素
    int Height;                  // 图像高度，像素；>0像素行倒存，<0正向存储
    unsigned short Planes;       // 色彩平面数，BMP固定等于1
    unsigned short BitCount;     // 每个像素位深；24代表24位真彩色
    unsigned int Compression;    // 压缩方式；0=不压缩
    unsigned int ImageSize;      // 像素数据总字节（包含行padding填充，可为0）
    unsigned int XPelsPerMeter;  // 水平分辨率：每米多少像素
    unsigned int YPelsPerMeter;  // 垂直分辨率：每米多少像素
    unsigned int ClrUsed;        // 使用的调色板颜色数量，24位真彩色为0
    unsigned int ClrImportant;   // 重要颜色数量，一般为0
};
typedef struct BITMAP_INFORMATION bitmapInfomation;

int main()
{
    FILE *bmpfile = fopen("D:/BC101/Resources/L09/640_480_RED.bmp", "rb");
    if (bmpfile == NULL)
    {
        printf("打开BMP文件失败\n");
        return 1;
    }
    printf("打开BMP文件成功\n");
    // 获取标识信息
    char tag[2];
    // fread(缓冲区,每个元素大小,元素数量,文件)
    size_t ret_1 = fread(tag, 2, 1, bmpfile);
    if (ret_1 != 1)
    {
        printf("读取BMP标识失败，文件过短或损坏\n");
        fclose(bmpfile);
        return 2;
    }
    char first = tag[0];
    char second = tag[1];
    printf("%c%c\n", first, second);
    if (first != 'B' || second != 'M')
    {
        printf("打开的文件不是BMP文件格式\n");
        fclose(bmpfile);
        return 3;
    }
    printf("读取BMP文件标识成功\n");

    // 开始处理位图文件图信息
    BitmapHeader bmpHeader;

    size_t ret_2 = fread(&bmpHeader, sizeof(bmpHeader), 1, bmpfile);
    if (ret_2 != 1)
    {
        printf("读取BMP位图文件头异常\n");
        fclose(bmpfile);
        return 4;
    }
    printf("BMP文件大小: %u\n", bmpHeader.Size);
    printf("像素偏移量: %u\n", bmpHeader.OffBits);

    // 开始处理位图详情信息
    bitmapInfomation bitInfo;
    size_t ret_3 = fread(&bitInfo, sizeof(bitInfo), 1, bmpfile);
    if (ret_3 != 1)
    {
        printf("读取位图详情信息失败\n");
        fclose(bmpfile);
        return 5;
    }

    printf("==== BITMAPINFOHEADER ====\n");
    printf("信息头大小: %u\n", bitInfo.InfoHeaderSize);
    printf("图像宽: %d\n", bitInfo.Width);
    printf("图像高: %d\n", bitInfo.Height);
    printf("色彩平面: %u\n", bitInfo.Planes);
    printf("位深: %u\n", bitInfo.BitCount);
    printf("压缩方式: %u (0=无压缩)\n", bitInfo.Compression);
    printf("像素数据总大小(含padding): %u\n", bitInfo.ImageSize);
    printf("X分辨率(每米像素): %u\n", bitInfo.XPelsPerMeter);
    printf("Y分辨率(每米像素): %u\n", bitInfo.YPelsPerMeter);
    printf("调色板颜色数量: %u\n", bitInfo.ClrUsed);
    printf("重要颜色数量: %u\n", bitInfo.ClrImportant);

    int bytes_per_pixel = bitInfo.BitCount / 8;           // 每个像素占多少字节，24位就是3字节
    int line_raw_bytes = bitInfo.Width * bytes_per_pixel; // 一行原始像素字节，不含padding填充
    int line_padded_bytes = (line_raw_bytes + 3) & (~3);  // 4字节对齐后的真实一行大小，包含padding
    printf("每行原始字节：%d, 对齐后实际行大小:%d\n", line_raw_bytes, line_padded_bytes);

    // 跳到像素数据真正起始位置，不要硬编码54
    fseek(bmpfile, bmpHeader.OffBits, SEEK_SET);
    printf("已经fseek跳到像素偏移：%u\n", bmpHeader.OffBits);

    printf("sizeof(BitmapHeader)=%zu\n", sizeof(bmpHeader));
    printf("sizeof(bitmapInformation)=%zu\n", sizeof(bitInfo));

    fclose(bmpfile);

    return 0;
}

为什么不定义一个结构体包含全部文件头的内容，然后一次性读入54个字节呢？

是由于在结构体进行内存分配时存在“内存对齐”因素，把BMP文件头的全部内容定义成一个结构体会导致从文件中读入的数据与结构体元素发生错位！

因此没有采用定义1个结构体存储所有文件头信息的方法。读取BMP文件头的主要目的是为了获取图片的关键信息，如宽度、高度和颜色数。在文件头读取完毕后，从第55个字节开始的内容会因为文件格式不同而分为两种情况：

  • 对于单色(1位)、16色(4位)、256色(8位)位图，接下来是调色板数据；
  • 对于16位、24位真彩色位图，接下来是像素数据。

为了减少程序的复杂度，我们只考虑24位位图的情况：

  • 从第55字节开始就是像素数据；
  • 1个像素的颜色使用3个字节来描述。
9.2.2 从文件中获取第1个像素的颜色

24位BMP文件每个像素占用3个字节，分别存储R/G/B颜色分量。

📝 注意
在BMP文件中，R、G、B颜色分量存储的次序是B、G、R，而不是我们习惯的次序-R、G、B。刚好是反的！

可以看到从第55字节开始的像素数据是00、00、 ff，表示这个像素的颜色分量值是0（B）、0（G）、255（R)。

为了便于存储读取的像素颜色，我们可以定义一个名为_24BIT_PIXEL的结构体，专门存储这种24位像素数据（之所以要在结构体和数据类型前面加下划线是因为C语言不支持以数字开始的结构体和数据类型名）。这个结构体使用3个unsigned char类型的成员变量（1字节）依次存储B、G、R三种颜色分量的值。
